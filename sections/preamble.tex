\newcounter{qnum}

% Answer box: full-width box before each question
\newcommand{\answerbox}[1]{%
  \par\vspace{2pt}
  \fbox{\begin{minipage}[t][#1][t]{0.8\linewidth}\vspace{4pt}\end{minipage}}%
  \par\vspace{4pt}
}

% Question item: takes answer height, question text, marks, source
\newcommand{\qitem}[4]{%
  \refstepcounter{qnum}
  \noindent\textbf{Q\theqnum.} #1\\
  {\small \textit{Marks: #2 \hfill Source: #3}}\par\vspace{6pt}
  \answerbox{#4}%
}
